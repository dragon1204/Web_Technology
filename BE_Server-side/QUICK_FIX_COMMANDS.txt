# Bước 1: Tìm và vào thư mục project
find /home /var/www /root /opt -name "BE_Server-side" -type d 2>/dev/null
cd /root/BE_Server-side

# Bước 2: Pull code
git pull origin main

# Bước 3: Xóa folder sensor
rm -rf src/modules/sensor

# Bước 4: Xóa dist và cache
rm -rf dist node_modules/.cache

# Bước 5: Generate Prisma
npx prisma generate

# Bước 6: Build
npm run build

# Bước 7: Restart
pm2 restart all
pm2 logs --lines 50
